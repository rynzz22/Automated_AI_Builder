    class AudioEngine {
        constructor() {
            this.ctx = null;
            this.isMuted = false;
        }

        init() {
            if (!this.ctx) {
                const AudioCtx = window.AudioContext || window.webkitAudioContext;
                if (AudioCtx) {
                    this.ctx = new AudioCtx();
                }
            }
            if (this.ctx && this.ctx.state === 'suspended') {

            }

        playChime() {
            if (this.isMuted) return;
            this.init();
            if (!this.ctx) return;

            const now = this.ctx.currentTime;
            
            // Harmonious 3-tone chime sequence (C5, E5, G5, C6)
            const notes = [523.25, 659.25, 783.99, 1046.50];
            
            notes.forEach((freq, index) => {
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();

                osc.type = 'sine';
                osc.frequency.setValueAtTime(freq, now + index * 0.12);

                gain.gain.setValueAtTime(0.01, now + index * 0.12);
                gain.gain.exponentialRampToValueAtTime(0.3, now + index * 0.12 + 0.02);
                gain.gain.exponentialRampToValueAtTime(0.0001, now + index * 0.12 + 0.6);

                osc.connect(gain);
                gain.connect(this.ctx.destination);

                osc.start(now + index * 0.12);
                osc.stop(now + index * 0.12 + 0.65);
            });
        }

    function renderCountdownDisplay() {
        if (state.countdown.status === 'running') {
            state.countdown.remainingMs = Math.max(0, state.countdown.targetTime - Date.now());
        }

        const remaining = state.countdown.remainingMs;
        DOM.cdDisplayTime.textContent = formatCountdownTime(remaining);

        // Progress percentage calculation with full floating point accuracy
        const percentage = state.countdown.totalMs > 0 
            ? Math.max(0, Math.min(100, (remaining / state.countdown.totalMs) * 100))
            : 0;
        
        DOM.cdProgressBar.style.width = `${percentage}%`;

        // Check for Timer Completion
        if (remaining <= 0 && state.countdown.status === 'running') {
            handleCountdownCompletion();
        }
    }

    function handleCountdownCompletion() {
        state.countdown.status = 'stopped';
        audio.playChime();

        document.title = '[ALARM] TIME\'S UP! - Chrono';
        updateFavicon('#ef4444');

        DOM.cdDisplayTime.textContent = '00:00';
        DOM.cdProgressBar.style.width = '0%';
        DOM.cdStartBtn.textContent = 'Start';
        DOM.cdStartBtn.classList.replace('btn-secondary', 'btn-primary');

        // Flash error/completion alert ring
        DOM.cdActiveContainer.classList.add('shake-error');
        setTimeout(() => {
            DOM.cdActiveContainer.classList.remove('shake-error');
        }, 1000);
    }

    // =========================================================================
    // 9. EVENT LISTENERS & INPUT SANITIZATION
    // =========================================================================
    function setupEventListeners() {
        // Tab Navigation
        DOM.tabBtns.forEach(btn => {
            btn.addEventListener('click', () => {
                const targetTab = btn.getAttribute('data-tab');

                DOM.tabBtns.forEach(b => {
                    b.classList.remove('active');
                    b.setAttribute('aria-selected', 'false');
                });
                DOM.tabPanels.forEach(p => p.classList.add('hidden'));

                btn.classList.add('active');
                btn.setAttribute('aria-selected', 'true');

                const activePanel = document.getElementById(`panel-${targetTab}`);
                if (activePanel) activePanel.classList.remove('hidden');

                state.activeTab = targetTab;
                saveState();
            });
        });

        // Theme Switcher
        DOM.themeToggleBtn.addEventListener('click', () => {
            state.theme = state.theme === 'dark' ? 'light' : 'dark';
            DOM.html.setAttribute('data-theme', state.theme);
            
            if (state.theme === 'dark') {
                DOM.themeDarkIcon.classList.remove('hidden');
                DOM.themeLightIcon.classList.add('hidden');
            } else {
                DOM.themeDarkIcon.classList.add('hidden');
                DOM.themeLightIcon.classList.remove('hidden');
            }
            saveState();
        });

        // Sound Switcher
        DOM.soundToggleBtn.addEventListener('click', () => {
            state.isMuted = !state.isMuted;
            audio.setMuted(state.isMuted);

            if (state.isMuted) {
                DOM.soundOnIcon.classList.add('hidden');
                DOM.soundOffIcon.classList.remove('hidden');
            } else {
                DOM.soundOnIcon.classList.remove('hidden');
                DOM.soundOffIcon.classList.add('hidden');
            }
            saveState();
        });

        // Audio Test Button
        DOM.audioTestBtn.addEventListener('click', () => {
            audio.playChime();
        });

        // Clock Controls
        DOM.clockFormatToggle.addEventListener('click', () => {
            state.clock24h = !state.clock24h;
            DOM.clockFormatToggle.textContent = state.clock24h ? '24H' : '12H';
            updateClock();
            saveState();
        });

        // Stopwatch Controls
        DOM.swStartBtn.addEventListener('click', () => {
            if (state.stopwatch.status === 'running') {
                pauseStopwatch();
            } else {
                startStopwatch();
            }
        });

        DOM.swLapBtn.addEventListener('click', recordLap);
        DOM.swResetBtn.addEventListener('click', resetStopwatch);
        DOM.swExportBtn.addEventListener('click', exportLapsCSV);
        DOM.swClearLapsBtn.addEventListener('click', clearLaps);

        // Countdown Controls
        DOM.cdStartBtn.addEventListener('click', () => {
            if (state.countdown.status === 'running') {
                pauseCountdown();
            } else {
                startCountdown();
            }
        });

        DOM.cdResetBtn.addEventListener('click', resetCountdown);
        DOM.cdAdd1m.addEventListener('click', () => addCountdownTime(60));
        DOM.cdAdd5m.addEventListener('click', () => addCountdownTime(300));

        // Countdown Presets
        DOM.cdPresets.querySelectorAll('.preset-btn').forEach(btn => {
            btn.addEventListener('click', () => {
                const secs = parseInt(btn.getAttribute('data-seconds'), 10) || 0;
                const h = Math.floor(secs / 3600);
                const m = Math.floor((secs % 3600) / 60);
                const s = secs % 60;

                DOM.cdHours.value = padZero(h);
                DOM.cdMinutes.value = padZero(m);
                DOM.cdSeconds.value = padZero(s);

                DOM.cdErrorBadge.classList.add('hidden');
            });
        });

        // Input Field Corruption Protection (Strict Key Filter)
        [DOM.cdHours, DOM.cdMinutes, DOM.cdSeconds].forEach(input => {
            input.addEventListener('keydown', (e) => {
                // Reject invalid character entries: 'e', 'E', '+', '-', '.', ','
                if (['e', 'E', '+', '-', '.', ','].includes(e.key)) {
                    e.preventDefault();
                }
            });

            input.addEventListener('blur', () => {
                sanitizeCountdownInputs();
            });

            input.addEventListener('input', () => {
                DOM.cdErrorBadge.classList.add('hidden');
            });
        });
    }

    // =========================================================================
    // 10. MAIN UNIFIED ENGINE LOOP (Worker Driven)
    // =========================================================================
    function initEngineLoop() {
        precisionWorker.onmessage = function (e) {
            if (e.data.type === 'tick') {
                // Always update clock display
                updateClock();

                // Update stopwatch if active
                if (state.stopwatch.status === 'running') {
                    renderStopwatchDisplay();
                }

                // Update countdown if active
                if (state.countdown.status === 'running') {
                    renderCountdownDisplay();
                }

                // Synchronize dynamic tab titles & dynamic favicons
                updateTabTitle();
            }
        };

        // Command precision worker to post tick messages every 25ms
        precisionWorker.postMessage({ command: 'start', interval: 25 });
    }

    // =========================================================================
    // 11. INITIALIZATION & RECOVERY
    // =========================================================================
    function init() {
        loadState();

        // Apply saved theme
        DOM.html.setAttribute('data-theme', state.theme);
        if (state.theme === 'light') {
            DOM.themeDarkIcon.classList.add('hidden');
            DOM.themeLightIcon.classList.remove('hidden');
        }

        // Apply saved sound setting
        audio.setMuted(state.isMuted);
        if (state.isMuted) {
            DOM.soundOnIcon.classList.add('hidden');
            DOM.soundOffIcon.classList.remove('hidden');
        }

        // Apply saved clock format
        DOM.clockFormatToggle.textContent = state.clock24h ? '24H' : '12H';

        // Apply saved tab
        const activeTabBtn = document.querySelector(`.tab-btn[data-tab="${state.activeTab}"]`);
        if (activeTabBtn) {
            activeTabBtn.click();
        }

        setupEventListeners();
        updateClock();
        initEngineLoop();
    }

    // Run application on DOM ready
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', init);
    } else {
        init();
    }
})();
